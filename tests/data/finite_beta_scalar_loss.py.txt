def objective_rows(self,state,u):
    rt=im.runtime_from_params(self.params(u),self.solver.implicit)
    surf=surfacerzfourier_from_boundary(*self.boundary(state),self.inp.nfp,nphi=c.NPHI,ntheta=c.NTHETA)
    coils=self.coils(u)
    bn,weights=self.normal_rows(state,u)
    smooth=jax.scipy.special.logsumexp(2000*jnp.sqrt(bn**2+1e-12))/2000
    value=(0.5*self.qs.total_state(state,rt)+0.5*c.ASPECT_WEIGHT*(opt.aspect_ratio(state,rt)-c.ASPECT_TARGET)**2
        +0.5*c.IOTA_WEIGHT*jnp.maximum(c.IOTA_FLOOR-opt.min_abs_iota(state,rt),0)**2
        +0.5*c.NORMAL_FIELD_WEIGHT*jnp.sum(weights*bn**2)
        +0.5*c.NORMAL_FIELD_LIMIT_WEIGHT*jnp.maximum(smooth-c.NORMAL_FIELD_OBJECTIVE_LIMIT,0)**2
        +0.5*c.LENGTH_WEIGHT*jnp.sum((coils.length[:3]-c.LENGTH_TARGET)**2)
        +0.5*c.CURVATURE_WEIGHT*jnp.sum(jnp.maximum(coils.curvature[:3]-c.CURVATURE_OBJECTIVE_LIMIT,0)**2)
        +0.5*c.COIL_DISTANCE_WEIGHT*loss_coil_separation(coils,c.COIL_DISTANCE_LIMIT,block_size=32)
        +0.5*c.COIL_SURFACE_DISTANCE_WEIGHT*loss_coil_surface_distance(coils,surf,c.COIL_SURFACE_DISTANCE_LIMIT,block_size=32))
    physical=c.physical_values(state,rt)
    return optimizer_rows(value,physical,getattr(self.args,'fixed_pressure',False))
